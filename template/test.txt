<script>
// Village Filters
function filterVillageDistricts() {
    const stateId = document.getElementById("villageState").value;
    const districtSelect = document.getElementById("villageDistrict");
    districtSelect.value = '';
    for(let i=0; i<districtSelect.options.length; i++){
        let opt = districtSelect.options[i];
        if(!opt.value) continue;
        opt.style.display = (opt.dataset.state === stateId) ? 'block' : 'none';
    }
    filterVillages();
}

function filterVillages() {
    const districtId = document.getElementById("villageDistrict").value;
    const villageSelect = document.getElementById("village");
    villageSelect.value = '';
    for(let i=0; i<villageSelect.options.length; i++){
        let opt = villageSelect.options[i];
        if(!opt.value) continue;
        opt.style.display = (opt.dataset.district === districtId) ? 'block' : 'none';
    }
}
